# ============================================================================
#  AI Review — 自动 AI 代码审阅
# ----------------------------------------------------------------------------
#  触发条件:
#    pull_request: opened / reopened / synchronize
#    workflow_dispatch: 手动跑某个 PR（填 PR 编号）
#
#  只审阅指定作者的 PR。作者名单来自变量 AI_REVIEW_AUTHORS（必须是 JSON 数组），
#  未设置时默认 ["vshengbro"]。要把这个 workflow 推给所有 PR 看，把下面 jobs.review.if
#  里的 fromJSON(...) 判断删掉即可。
#
#  需要配置的 Secret（二选一，都可加；auto 模式下优先 MiniMax）:
#    MINIMAX_API_KEY  -> 用 MiniMax 模型审阅（OpenAI 兼容接口）
#    COPILOT_TOKEN    -> 改用 GitHub Copilot 原生审阅（PAT，需要 pull_requests: write，
#                        且该账号得有 Copilot code review 权限）
#
#  可选 Variable（组织级或仓库级均可）:
#    AI_REVIEW_PROVIDER       minimax | copilot | auto   （默认 auto）
#    AI_REVIEW_MODEL          默认 MiniMax-M2.5
#    AI_REVIEW_API_BASE       默认 https://api.minimax.io/v1
#    AI_REVIEW_AUTHORS        JSON 数组，默认 ["vshengbro"]
#    AI_REVIEW_MAX_DIFF_CHARS 单次送模型的 diff 上限，默认 120000
#    AI_REVIEW_MAX_TOKENS     模型输出上限，默认 6000
#    AI_REVIEW_MAX_FINDINGS   最多提多少条建议，默认 20
#    AI_REVIEW_EXCLUDE        逗号分隔的忽略规则，默认见脚本内
#
#  产物: 每条问题作为行级 review comment 贴在对应代码行上，另加一条总结 review。
# ============================================================================
name: AI Review

on:
  pull_request:
    types: [opened, reopened, synchronize]
  workflow_dispatch:
    inputs:
      pr_number:
        description: "要审阅的 PR 编号（手动触发时填）"
        required: true

permissions:
  contents: read
  pull-requests: write

concurrency:
  group: ai-review-${{ github.event.pull_request.number || inputs.pr_number }}
  cancel-in-progress: true

jobs:
  review:
    if: >-
      ${{ github.event_name == 'workflow_dispatch'
          || contains(fromJSON(vars.AI_REVIEW_AUTHORS || '["vshengbro"]'),
                      github.event.pull_request.user.login) }}
    runs-on: ubuntu-latest
    timeout-minutes: 25
    env:
      GITHUB_TOKEN: ${{ secrets.GITHUB_TOKEN }}
      PR_NUMBER: ${{ github.event.pull_request.number || inputs.pr_number }}
      MINIMAX_API_KEY: ${{ secrets.MINIMAX_API_KEY }}
      COPILOT_TOKEN: ${{ secrets.COPILOT_TOKEN }}
      AI_REVIEW_PROVIDER: ${{ vars.AI_REVIEW_PROVIDER }}
      AI_REVIEW_MODEL: ${{ vars.AI_REVIEW_MODEL }}
      AI_REVIEW_API_BASE: ${{ vars.AI_REVIEW_API_BASE }}
      AI_REVIEW_MAX_DIFF_CHARS: ${{ vars.AI_REVIEW_MAX_DIFF_CHARS }}
      AI_REVIEW_MAX_TOKENS: ${{ vars.AI_REVIEW_MAX_TOKENS }}
      AI_REVIEW_MAX_FINDINGS: ${{ vars.AI_REVIEW_MAX_FINDINGS }}
      AI_REVIEW_EXCLUDE: ${{ vars.AI_REVIEW_EXCLUDE }}
    steps:
      - name: 选后端
        id: provider
        shell: bash
        run: |
          set -euo pipefail
          p="${AI_REVIEW_PROVIDER:-auto}"
          if [ -z "$p" ] || [ "$p" = "auto" ]; then
            if [ -n "${MINIMAX_API_KEY:-}" ]; then
              p="minimax"
            elif [ -n "${COPILOT_TOKEN:-}" ]; then
              p="copilot"
            else
              p="none"
            fi
          fi
          case "$p" in
            minimax)
              if [ -z "${MINIMAX_API_KEY:-}" ]; then
                echo "::error::AI_REVIEW_PROVIDER=minimax，但没有配置 secret MINIMAX_API_KEY"
                exit 1
              fi
              ;;
            copilot)
              if [ -z "${COPILOT_TOKEN:-}" ]; then
                echo "::error::AI_REVIEW_PROVIDER=copilot，但没有配置 secret COPILOT_TOKEN"
                exit 1
              fi
              ;;
            none)
              echo "::notice::没有配置 MINIMAX_API_KEY / COPILOT_TOKEN，本次跳过 AI 审阅"
              exit 0
              ;;
            *)
              echo "::error::AI_REVIEW_PROVIDER 只能是 minimax / copilot / auto，收到 '$p'"
              exit 1
              ;;
          esac
          echo "provider=$p" >> "$GITHUB_OUTPUT"

      - name: 请求 GitHub Copilot 审阅
        if: steps.provider.outputs.provider == 'copilot'
        shell: bash
        run: |
          set -euo pipefail
          out=$(curl -sS -w '\n%{http_code}' -X POST \
            -H "Authorization: Bearer $COPILOT_TOKEN" \
            -H "Accept: application/vnd.github+json" \
            -H "X-GitHub-Api-Version: 2022-11-28" \
            -H "Content-Type: application/json" \
            "https://api.github.com/repos/$GITHUB_REPOSITORY/pulls/$PR_NUMBER/requested_reviewers" \
            -d '{"reviewers":["copilot-pull-request-reviewer[bot]"]}')
          code=$(printf '%s' "$out" | tail -n1)
          body=$(printf '%s' "$out" | sed '$d')
          if [ "$code" -ge 200 ] && [ "$code" -lt 300 ]; then
            echo "::notice::已请求 Copilot 审阅 PR #$PR_NUMBER，审阅结果稍后出现在 PR 上"
            printf '%s\n' "$body" | head -c 500
          else
            echo "::warning::请求 Copilot 审阅失败 HTTP $code: $body"
            exit 1
          fi

      - name: MiniMax AI 审阅
        if: steps.provider.outputs.provider == 'minimax'
        shell: bash
        run: |
          python3 - <<'PYEOF'
          #!/usr/bin/env python3
          """
          AI PR Review — self-contained review bot for GitHub Actions.
          
          Runs inside `pull_request` workflows. It:
            1. reads the PR diff through the GitHub REST API (no checkout needed),
            2. asks a chat model for a structured review,
            3. posts each finding as an inline review comment,
            4. posts one summary review for the whole PR.
          
          Every network call is isolated: a single rejected inline comment never aborts
          the run, and the summary is always posted (review API first, issue comment as
          fallback).
          """
          
          import fnmatch
          import json
          import os
          import re
          import sys
          import urllib.error
          import urllib.request
          
          # --------------------------------------------------------------------------- #
          # configuration
          # --------------------------------------------------------------------------- #
          
          GITHUB_TOKEN = os.environ.get("GITHUB_TOKEN", "")
          REPO = os.environ.get("GITHUB_REPOSITORY", "")
          PR_NUMBER = os.environ.get("PR_NUMBER", "")
          
          API = "https://api.github.com"
          MODEL = os.environ.get("AI_REVIEW_MODEL") or "MiniMax-M2.5"
          API_BASE = (os.environ.get("AI_REVIEW_API_BASE") or "https://api.minimax.io/v1").rstrip("/")
          MODEL_KEY = os.environ.get("MINIMAX_API_KEY", "")
          
          MAX_DIFF_CHARS = int(os.environ.get("AI_REVIEW_MAX_DIFF_CHARS") or 120000)
          MAX_TOKENS = int(os.environ.get("AI_REVIEW_MAX_TOKENS") or 6000)
          MAX_FINDINGS = int(os.environ.get("AI_REVIEW_MAX_FINDINGS") or 20)
          
          EXCLUDE = [
              p.strip()
              for p in (os.environ.get("AI_REVIEW_EXCLUDE")
                        or "*.lock,package-lock.json,pnpm-lock.yaml,yarn.lock,dist/**,target/**,node_modules/**,*.min.js,*.map")
              .split(",")
              if p.strip()
          ]
          
          SEVERITY_ORDER = {"high": 0, "medium": 1, "low": 2}
          SEVERITY_EMOJI = {"high": "🔴", "medium": "🟡", "low": "🔵"}
          SEVERITY_LABEL = {"high": "严重", "medium": "中等", "low": "提示"}
          
          
          def log(msg):
              print(msg, flush=True)
          
          
          def notice(msg):
              print(f"::notice::{msg}", flush=True)
          
          
          def warn(msg):
              print(f"::warning::{msg}", flush=True)
          
          
          # --------------------------------------------------------------------------- #
          # small helpers
          # --------------------------------------------------------------------------- #
          
          def is_excluded(path):
              for pat in EXCLUDE:
                  if pat.endswith("/**"):
                      if path.startswith(pat[:-2]):
                          return True
                  elif fnmatch.fnmatch(path, pat):
                      return True
              return False
          
          
          def api(method, path, body=None, accept="application/vnd.github+json", raw=False, timeout=90):
              """Call the GitHub REST API. Raises RuntimeError with a readable message."""
              url = path if path.startswith("http") else API + path
              data = json.dumps(body).encode("utf-8") if body is not None else None
              req = urllib.request.Request(url, data=data, method=method)
              req.add_header("Authorization", f"Bearer {GITHUB_TOKEN}")
              req.add_header("Accept", accept)
              req.add_header("X-GitHub-Api-Version", "2022-11-28")
              req.add_header("User-Agent", "ai-review-action")
              if data is not None:
                  req.add_header("Content-Type", "application/json")
              try:
                  with urllib.request.urlopen(req, timeout=timeout) as resp:
                      payload = resp.read()
                      if raw:
                          return payload.decode("utf-8", "replace")
                      return json.loads(payload) if payload else {}
              except urllib.error.HTTPError as exc:
                  detail = exc.read().decode("utf-8", "replace")
                  raise RuntimeError(f"{method} {path} -> HTTP {exc.code}: {detail[:400]}")
              except urllib.error.URLError as exc:
                  raise RuntimeError(f"{method} {path} -> network error: {exc.reason}")
          
          
          def extract_json(text):
              """Pull a JSON object out of a model response, tolerating prose and fences."""
              if not text:
                  return None
              text = re.sub(r"<think>.*?</think>", "", text, flags=re.S)
              # try every fenced block first, then the raw text
              candidates = [m.group(1) for m in re.finditer(r"```(?:json)?\s*(.*?)```", text, re.S)]
              candidates.append(text)
              for cand in candidates:
                  start = cand.find("{")
                  end = cand.rfind("}")
                  if start == -1 or end <= start:
                      continue
                  blob = cand[start:end + 1]
                  for attempt in (blob, re.sub(r",\s*([}\]])", r"\1", blob)):
                      try:
                          return json.loads(attempt)
                      except ValueError:
                          continue
              return None
          
          
          HUNK_RE = re.compile(r"^@@ -(\d+)(?:,(\d+))? \+(\d+)(?:,(\d+))? @@")
          
          
          def parse_hunks(patch):
              """Map a unified diff to the set of new-file line numbers the diff covers.
          
              Returns (covered, added): line numbers that exist in the new file and are
              part of some hunk, and the subset of those that are newly added lines.
              Only these lines are accepted by the inline-comment API.
              """
              covered, added = set(), set()
              new_line = None
              hunk_end = None
              for raw in patch.split("\n"):
                  head = HUNK_RE.match(raw)
                  if head:
                      start = int(head.group(3))
                      count = int(head.group(4)) if head.group(4) is not None else 1
                      new_line = start
                      hunk_end = start + count - 1
                      covered.update(range(start, hunk_end + 1))
                      continue
                  if new_line is None:
                      continue
                  if raw.startswith("\\"):
                      continue  # "\ No newline at end of file"
                  if raw.startswith("+"):
                      added.add(new_line)
                      covered.add(new_line)
                      new_line += 1
                  elif raw.startswith("-"):
                      continue  # removed line, has no number on the new side
                  elif raw.startswith(" ") or raw == "":
                      covered.add(new_line)
                      new_line += 1
                  else:
                      new_line = None
                      hunk_end = None
                      continue
                  if hunk_end is not None and new_line > hunk_end:
                      new_line = None
                      hunk_end = None
              return covered, added
          
          
          def severity_of(value):
              value = str(value or "medium").strip().lower()
              if value in ("high", "critical", "blocker", "p0", "p1", "严重", "高"):
                  return "high"
              if value in ("low", "nit", "info", "minor", "提示", "低"):
                  return "low"
              return "medium"
          
          
          def language_hint(paths):
              exts = {
                  ".rs": "Rust", ".ts": "TypeScript", ".tsx": "TypeScript/React",
                  ".js": "JavaScript", ".jsx": "JavaScript/React", ".py": "Python",
                  ".go": "Go", ".java": "Java", ".kt": "Kotlin", ".swift": "Swift",
                  ".c": "C", ".h": "C", ".cpp": "C++", ".cs": "C#", ".rb": "Ruby",
                  ".php": "PHP", ".sh": "Shell", ".yml": "YAML", ".yaml": "YAML",
                  ".json": "JSON", ".toml": "TOML", ".md": "Markdown", ".sql": "SQL",
              }
              langs = {exts[p[p.rfind("."):].lower()] for p in paths if "." in p and p.rfind(".") > 0}
              langs.discard(None)
              return ", ".join(sorted(langs)) or "unknown"
          
          
          SYSTEM_PROMPT = """\
          You are a meticulous senior software engineer reviewing a pull request.
          
          Your job is to find real defects and meaningful risks in the diff, not to
          nitpick formatting. Prioritise, in order:
            1. correctness bugs, logic errors, off-by-one, wrong units, bad edge cases
            2. security: injection, authz/authn gaps, secret handling, unsafe deserialisation
            3. concurrency and resource management: races, deadlocks, leaks, missing cleanup
            4. error handling: swallowed errors, wrong error types, missing propagation
            5. breaking API / schema / behaviour changes that are not reflected in the diff
            6. missing or wrong tests for the changed behaviour
          
          Rules:
            - Only report problems you can point at in the diff. Do not speculate about
              code you cannot see.
            - Skip pure style, formatting, import order, comment wording and typo noise.
            - If the diff looks correct, return an empty findings array. That is a valid
              and useful answer — do not invent issues.
            - `line` must be a line number in the NEW file (the `+` side of the diff),
              and it must be a line that appears in the diff. Use the exact file path as
              it appears in the diff headers.
            - `suggestion` is optional: a short concrete replacement snippet or fix
              description. Use an empty string when you have nothing to propose.
          
          Respond with a single JSON object and nothing else. No markdown fences, no
          prose before or after. Use exactly this shape:
          
          {
            "verdict": "approve" | "comment" | "request_changes",
            "summary": "two to four sentences describing what the change does and your overall read",
            "findings": [
              {
                "file": "path/from/the/diff",
                "line": 42,
                "severity": "high" | "medium" | "low",
                "title": "short imperative title",
                "detail": "what is wrong, why it matters, and the concrete failure it causes",
                "suggestion": "optional fix snippet or instruction"
              }
            ]
          }
          """
          
          
          def build_user_prompt(pr, files, diff_text, truncated):
              meta = {
                  "repository": REPO,
                  "title": pr.get("title") or "",
                  "description": (pr.get("body") or "")[:4000],
                  "base_branch": (pr.get("base") or {}).get("ref") or "",
                  "head_branch": (pr.get("head") or {}).get("ref") or "",
                  "languages": language_hint([f["filename"] for f in files]),
                  "changed_files": [
                      {
                          "path": f["filename"],
                          "status": f.get("status"),
                          "additions": f.get("additions"),
                          "deletions": f.get("deletions"),
                      }
                      for f in files
                  ],
              }
              note = (
                  "\n\nNOTE: the diff below was truncated because it is large. "
                  "Review only the parts you can see, and say so in the summary."
                  if truncated else ""
              )
              return (
                  f"Review this pull request.\n\n"
                  f"=== PR METADATA ===\n{json.dumps(meta, ensure_ascii=False, indent=2)}\n\n"
                  f"=== UNIFIED DIFF ===\n{diff_text}{note}\n\n"
                  f"Return the JSON object described in your instructions."
              )
          
          
          def call_model(prompt):
              payload = {
                  "model": MODEL,
                  "messages": [
                      {"role": "system", "content": SYSTEM_PROMPT},
                      {"role": "user", "content": prompt},
                  ],
                  "temperature": 0.2,
                  "max_tokens": MAX_TOKENS,
              }
              req = urllib.request.Request(
                  f"{API_BASE}/chat/completions",
                  data=json.dumps(payload).encode("utf-8"),
                  method="POST",
              )
              req.add_header("Authorization", f"Bearer {MODEL_KEY}")
              req.add_header("Content-Type", "application/json")
              req.add_header("User-Agent", "ai-review-action")
              try:
                  with urllib.request.urlopen(req, timeout=600) as resp:
                      body = json.loads(resp.read())
              except urllib.error.HTTPError as exc:
                  detail = exc.read().decode("utf-8", "replace")
                  raise RuntimeError(f"model API -> HTTP {exc.code}: {detail[:400]}")
              except urllib.error.URLError as exc:
                  raise RuntimeError(f"model API -> network error: {exc.reason}")
          
              choices = body.get("choices") or []
              if not choices:
                  raise RuntimeError(f"model API returned no choices: {json.dumps(body)[:300]}")
              content = (choices[0].get("message") or {}).get("content") or ""
              if isinstance(content, list):
                  content = "".join(part.get("text", "") for part in content if isinstance(part, dict))
              if not content.strip():
                  raise RuntimeError("model API returned empty content")
              return content
          
          
          def existing_keys(pr_number):
              """(path, line, title) triples already commented on, to avoid duplicates."""
              keys = set()
              page = 1
              while page <= 5:
                  try:
                      batch = api("GET", f"/repos/{REPO}/pulls/{pr_number}/comments?per_page=100&page={page}")
                  except RuntimeError:
                      break
                  if not batch:
                      break
                  for item in batch:
                      body = (item.get("body") or "")
                      marker = re.search(r"<!-- ai-review-key:(.+?) -->", body)
                      if marker:
                          keys.add(marker.group(1).strip())
                  if len(batch) < 100:
                      break
                  page += 1
              return keys
          
          
          def render_finding(finding):
              sev = severity_of(finding.get("severity"))
              title = str(finding.get("title") or "Issue").strip()
              detail = str(finding.get("detail") or "").strip()
              suggestion = str(finding.get("suggestion") or "").strip()
              parts = [f"**{SEVERITY_EMOJI[sev]} {SEVERITY_LABEL[sev]} · {title}**", ""]
              if detail:
                  parts += [detail, ""]
              if suggestion:
                  parts += ["**建议**", "", "```suggestion", suggestion, "```", ""]
              return "\n".join(parts).rstrip()
          
          
          def render_summary(verdict, summary, findings, skipped):
              emoji = {"approve": "✅", "comment": "💬", "request_changes": "🚨"}.get(verdict, "💬")
              label = {
                  "approve": "看起来可以合入",
                  "comment": "有建议但不阻塞",
                  "request_changes": "建议先改动再合入",
              }.get(verdict, "有建议")
              counts = {s: sum(1 for f in findings if f["severity"] == s) for s in ("high", "medium", "low")}
              out = [f"## {emoji} AI 审阅 · {label}", ""]
              if summary:
                  out += [summary, ""]
              out += [
                  f"共 {len(findings)} 条建议（🔴 严重 {counts['high']} · 🟡 中等 {counts['medium']} · 🔵 提示 {counts['low']}）",
                  "",
              ]
              if findings:
                  out += ["| 位置 | 级别 | 问题 |", "| --- | --- | --- |"]
                  for f in findings:
                      where = f"`{f['file']}:{f['line']}`" if f.get("line") else f"`{f['file']}`"
                      title = str(f.get("title") or "").replace("|", "\\|").replace("\n", " ")
                      out.append(f"| {where} | {SEVERITY_EMOJI[f['severity']]} {SEVERITY_LABEL[f['severity']]} | {title} |")
                  out.append("")
              else:
                  out += ["没有发现需要修改的问题。", ""]
              if skipped:
                  out += [f"> {len(skipped)} 条建议因为行号无法定位，只保留在上表/正文中。", ""]
              out += ["---", f"<sub>🤖 由 `{MODEL}` 自动生成，仅供参考，最终以人工评审为准。</sub>"]
              return "\n".join(out)
          
          
          # --------------------------------------------------------------------------- #
          # main
          # --------------------------------------------------------------------------- #
          
          def main():
              if not GITHUB_TOKEN or not REPO or not PR_NUMBER:
                  log("missing GITHUB_TOKEN / GITHUB_REPOSITORY / PR_NUMBER")
                  return 1
          
              pr = api("GET", f"/repos/{REPO}/pulls/{PR_NUMBER}")
              head_sha = (pr.get("head") or {}).get("sha")
          
              files, page = [], 1
              while page <= 10:
                  batch = api("GET", f"/repos/{REPO}/pulls/{PR_NUMBER}/files?per_page=100&page={page}")
                  if not batch:
                      break
                  files.extend(batch)
                  if len(batch) < 100:
                      break
                  page += 1
          
              if not files:
                  notice("no changed files, nothing to review")
                  return 0
          
              # build diff text + per-file line map
              diff_parts, line_map, skipped_files = [], {}, []
              for f in files:
                  path = f.get("filename") or ""
                  patch = f.get("patch") or ""
                  if is_excluded(path):
                      skipped_files.append(path)
                      continue
                  covered, added = parse_hunks(patch)
                  line_map[path] = {"covered": covered, "added": added}
                  header = f"--- a/{path}\n+++ b/{path}\n"
                  diff_parts.append(header + patch)
              diff_text = "\n".join(diff_parts)
          
              if not diff_text.strip():
                  notice("every changed file is excluded by AI_REVIEW_EXCLUDE, nothing to review")
                  return 0
          
              truncated = len(diff_text) > MAX_DIFF_CHARS
              if truncated:
                  diff_text = diff_text[:MAX_DIFF_CHARS] + "\n... [truncated]"
          
              log(f"files={len(files)} reviewed={len(line_map)} diff_chars={len(diff_text)} truncated={truncated}")
              if skipped_files:
                  log(f"excluded by pattern: {', '.join(skipped_files)}")
          
              prompt = build_user_prompt(pr, files, diff_text, truncated)
              raw = call_model(prompt)
              parsed = extract_json(raw)
              if not parsed:
                  warn("model did not return parsable JSON; posting the raw response instead")
                  api("POST", f"/repos/{REPO}/issues/{PR_NUMBER}/comments",
                      {"body": "## 🤖 AI 审阅\n\n模型输出无法解析为结构化结果，原始输出如下：\n\n```\n"
                               + raw[:8000] + "\n```"})
                  return 0
          
              verdict = str(parsed.get("verdict") or "comment").strip().lower()
              if verdict not in ("approve", "comment", "request_changes"):
                  verdict = "comment"
              summary = str(parsed.get("summary") or "").strip()
              raw_findings = parsed.get("findings")
              if not isinstance(raw_findings, list):
                  raw_findings = []
          
              # normalise + validate
              candidates, seen = [], set()
              for item in raw_findings[:MAX_FINDINGS * 2]:
                  if not isinstance(item, dict):
                      continue
                  path = str(item.get("file") or "").strip().lstrip("./")
                  if not path:
                      continue
                  if path not in line_map:
                      # tolerate a bare filename, but only when it identifies exactly one diff file
                      base = path.rsplit("/", 1)[-1]
                      matches = [p for p in line_map if p.rsplit("/", 1)[-1] == base]
                      if len(matches) != 1:
                          continue
                      path = matches[0]
                  try:
                      line = int(item.get("line"))
                  except (TypeError, ValueError):
                      line = None
                  if line is not None and line not in line_map[path]["covered"]:
                      line = None
                  title = str(item.get("title") or "Issue").strip()[:200]
                  key = f"{path}:{line}:{title.lower()}"
                  if key in seen:
                      continue
                  seen.add(key)
                  candidates.append({
                      "file": path,
                      "line": line,
                      "severity": severity_of(item.get("severity")),
                      "title": title,
                      "detail": str(item.get("detail") or "").strip(),
                      "suggestion": str(item.get("suggestion") or "").strip(),
                      "key": key,
                  })
          
              candidates.sort(key=lambda f: (SEVERITY_ORDER.get(f["severity"], 3), f["file"], f["line"] or 0))
              candidates = candidates[:MAX_FINDINGS]
          
              already = existing_keys(PR_NUMBER)
              posted, duplicated, unlocatable = [], [], []
          
              for f in candidates:
                  if f["key"] in already:
                      duplicated.append(f)
                      continue
                  if f["line"] is None:
                      unlocatable.append(f)
                      continue
                  body = render_finding(f) + f"\n\n<!-- ai-review-key:{f['key']} -->"
                  try:
                      api("POST", f"/repos/{REPO}/pulls/{PR_NUMBER}/comments", {
                          "commit_id": head_sha,
                          "path": f["file"],
                          "line": f["line"],
                          "side": "RIGHT",
                          "body": body,
                      })
                      posted.append(f)
                  except RuntimeError as exc:
                      warn(f"inline comment failed for {f['file']}:{f['line']} -> {exc}")
                      unlocatable.append(f)
          
              listed = posted + unlocatable
              summary_body = render_summary(verdict, summary, listed, unlocatable)
              try:
                  api("POST", f"/repos/{REPO}/pulls/{PR_NUMBER}/reviews", {
                      "commit_id": head_sha,
                      "body": summary_body,
                      "event": "COMMENT",
                  })
              except RuntimeError as exc:
                  warn(f"review API failed ({exc}); falling back to an issue comment")
                  try:
                      api("POST", f"/repos/{REPO}/issues/{PR_NUMBER}/comments", {"body": summary_body})
                  except RuntimeError as exc2:
                      log(f"could not post summary: {exc2}")
                      return 1
          
              if posted:
                  notice(f"AI review posted: {len(posted)} inline comment(s), verdict={verdict}")
              else:
                  notice(f"AI review posted: summary only, verdict={verdict}")
              if duplicated:
                  log(f"skipped {len(duplicated)} duplicate finding(s)")
          
              step_summary = os.environ.get("GITHUB_STEP_SUMMARY")
              if step_summary:
                  try:
                      with open(step_summary, "a", encoding="utf-8") as fh:
                          fh.write(summary_body + "\n")
                  except OSError:
                      pass
              return 0
          
          
          if __name__ == "__main__":
              try:
                  sys.exit(main())
              except RuntimeError as exc:
                  warn(str(exc))
                  sys.exit(1)
